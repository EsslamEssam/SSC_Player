<body>
    <div id="player"></div>
    <script>
        var tag = document.createElement('script');
        tag.src = "https://www.youtube.com/iframe_api";
        var firstScriptTag = document.getElementsByTagName('script')[0];
        firstScriptTag.parentNode.insertBefore(tag, firstScriptTag);

        var player;
        function loadVideo(videoId) {
            if (player && player.loadVideoById) {
                player.loadVideoById(videoId);
            } else {
                window.pendingVideoId = videoId;
            }
        }

        function onYouTubeIframeAPIReady() {
            player = new YT.Player('player', {
                videoId: window.pendingVideoId || '',
                playerVars: { 'controls': 0, 'modestbranding': 1, 'rel': 0, 'disablekb': 1, 'fs': 0 }
            });
        }

        function playVideo() { if(player) player.playVideo(); }
        function pauseVideo() { if(player) player.pauseVideo(); }
        function setVolume(val) { if(player) player.setVolume(val); }
        function seekTo(sec) { if(player) player.seekTo(sec, true); }
    </script>
